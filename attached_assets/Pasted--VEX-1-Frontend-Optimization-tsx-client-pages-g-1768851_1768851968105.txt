2. WebSocket Sessions Manager (مش موجود حالياً)
typescript
// server/services/gameSessions.ts
interface ActiveSession {
  id: string;
  gameId: string;
  players: Set<string>;  // player IDs
  startedAt: Date;
  maxPlayers: number;
}

const activeSessions = new Map<string, ActiveSession>();

// Clean expired sessions كل 30 ثانية
setInterval(() => {
  for (const [sessionId, session] of activeSessions) {
    if (Date.now() - session.startedAt.getTime() > 24 * 60 * 60 * 1000) {
      activeSessions.delete(sessionId);
    }
  }
}, 30000);
3. Database Connection Pooling (مهم جداً)
typescript
// server/lib/db.ts - Drizzle مع Pool
import { drizzle } from 'drizzle-orm/node-postgres';
import { Pool } from 'pg';

const pool = new Pool({
  connectionString: process.env.DATABASE_URL,
  max: 20,      // Max connections
  idleTimeoutMillis: 30000,
  connectionTimeoutMillis: 2000,
});

export const db = drizzle(pool);
4. Games Preloading Strategy
typescript
// client/hooks/useGames.ts
export const useGames = () => {
  return useQuery({
    queryKey: ['games'],
    queryFn: async () => {
      // Preload popular games أولاً
      const popular = await fetch('/api/games/popular?limit=8').then(r => r.json());
      const rest = await fetch('/api/games?limit=12').then(r => r.json());
      return [...popular, ...rest];
    },
    staleTime: 5 * 60 * 1000,  // 5 دقائق
  });
};
