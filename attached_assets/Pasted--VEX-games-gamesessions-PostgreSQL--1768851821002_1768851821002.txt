Express.js Implementation:

typescript
// server/routes/games.ts
import Redis from 'redis';

const redis = Redis.createClient();

app.get('/api/games', async (req, res) => {
  const { category, page = 1, limit = 20, featured } = req.query;
  const cacheKey = `games:${category}:${page}:${featured}`;
  
  // Cache 5 دقائق
  const cached = await redis.get(cacheKey);
  if (cached) return res.json(JSON.parse(cached));

  const offset = (Number(page) - 1) * Number(limit);
  const where: any = { status: 'active' };
  
  if (category) where.category = category;
  if (featured === 'true') where.featured = true;

  const games = await db.select()
    .from(games_metadata)
    .where(where)
    .limit(Number(limit))
    .offset(offset)
    .orderBy('priority', 'desc');

  await redis.setex(cacheKey, 300, JSON.stringify(games));
  res.json(games);
});
المرحلة 3: Frontend Infinite Scroll + Virtual List
React + TanStack Query + react-window:

tsx
// client/pages/games.tsx
import { useInfiniteQuery } from '@tanstack/react-query';
import { FixedSizeList as List } from 'react-window';

const GamesLobby = () => {
  const { data, fetchNextPage, hasNextPage, isFetchingNextPage } = useInfiniteQuery({
    queryKey: ['games'],
    queryFn: ({ pageParam = 1 }) => 
      fetch(`/api/games?page=${pageParam}&limit=20`).then(res => res.json()),
    getNextPageParam: (lastPage, pages) => 
      lastPage.length === 20 ? pages.length + 1 : undefined,
  });

  const Item = ({ index, style, data }: ListChildComponentProps<Game[]>) => (
    <div style={style}>
      <GameCard game={data[index % 20]} />
    </div>
  );

  return (
    <div className="games-lobby">
      <List
        height={600}
        itemCount={data?.pages.flat().length || 0}
        itemSize={200}
        itemData={data?.pages.flat()}
        width="100%"
      >
        {Item}
      </List>
      
      {hasNextPage && (
        <button 
          onClick={() => fetchNextPage()}
          disabled={isFetchingNextPage}
        >
          تحميل المزيد...
        </button>
      )}
    </div>
  );
};

Backend Game Session Manager:

typescript
// server/websocket/gameSessions.ts
interface GameSession {
  id: string;
  userId: string;
  gameId: string;
  provider: string;
  iframeUrl: string;
  sessionData: any;
}

const activeSessions: Map<string, GameSession> = new Map();

io.on('connection', (socket) => {
  socket.on('joinGame', async ({ gameId, userId }) => {
    const game = await db.select().from(games_metadata).where({ id: gameId });
    const sessionId = generateSessionId();
    
    // Dynamic iframe URL based on provider
    const iframeUrl = game[0].provider === 'custom' 
      ? '/games/custom/' + gameId 
      : `https://${game[0].providerUrl}/embed?session=${sessionId}&user=${userId}`;
    
    activeSessions.set(sessionId, { 
      id: sessionId, 
      userId, 
      gameId, 
      provider: game[0].provider,
      iframeUrl,
      sessionData: {}
    });

    socket.emit('gameReady', { sessionId, iframeUrl });
  });
});
المرحلة 5: Performance Monitoring + Limits

// server/middleware/rateLimiter.ts
import rateLimit from 'express-rate-limit';

const gameLimiter = rateLimit({
  windowMs: 1 * 60 * 1000, // 1 دقيقة
  max: 50, // 50 لعبة كحد أقصى في الدقيقة
  message: 'بطء في الطلبات، حاول مرة أخرى',
  standardHeaders: true,
  legacyHeaders: false,
});

app.use('/api/games', gameLimiter);
