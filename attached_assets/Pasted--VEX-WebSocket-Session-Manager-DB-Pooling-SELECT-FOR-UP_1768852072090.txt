// client/src/pages/games/GamesGrid.tsx - استبدل الـ Grid الحالي
import { FixedSizeGrid } from 'react-window';
import AutoSizer from 'react-virtualized-auto-sizer';

interface GameGridProps {
  games: Game[];
  onGameSelect: (game: Game) => void;
}

export const GamesGrid = ({ games, onGameSelect }: GameGridProps) => (
  <AutoSizer>
    {({ height, width }) => (
      <FixedSizeGrid
        columnCount={4}
        columnWidth={320}
        height={height}
        rowCount={Math.ceil(games.length / 4)}
        rowHeight={220}
        width={width}
        itemData={{ games, onGameSelect }}
      >
        {GridItem}
      </FixedSizeGrid>
    )}
  </AutoSizer>
);

const GridItem = ({ data, columnIndex, rowIndex, style }: any) => {
  const { games, onGameSelect } = data;
  const index = rowIndex * 4 + columnIndex;
  
  if (!games[index]) return null;
  
  return (
    <div style={style} className="p-2">
      <GameCard 
        game={games[index]} 
        onClick={() => onGameSelect(games[index])}
      />
    </div>
  );
};
